Vấn đề chính

🗄️ Database: SQLite
  • Không có connection pooling
  • Lock toàn database khi write
  • Giới hạn ~100K records hiệu quả
  • Không phù hợp production
🔍 Không có Indexes
  • Full table scan mỗi query
  • O(n) complexity cho tìm kiếm
  • Chậm dần theo số lượng records
  • CPU spike khi load nhiều
💾 Không có Caching
  • Mỗi request đều query DB
  • Duplicate queries không cần thiết
  • Tải database cao
  • Response time chậm
📄 Không có Pagination
  • Load tất cả records cùng lúc
  • Memory overflow với data lớn
  • Slow page load
  • Tốn băng thông không cần thiết
🔎 Không có Search Engine
  • Không thể full-text search
  • LIKE queries cực chậm
  • Không có ranking/relevance
  • Không scale với triệu records
⚡ ORM Queries không tối ưu
  • N+1 query problems
  • Select * thay vì specific columns
  • Lazy loading gây nhiều queries
  • Không có query optimization

1.3 Query Optimization

Chỉ select columns cần thiết, eager loading để tránh N+1

# Bad - Select all columns, lazy loading
articles = Article.query.filter_by(status='published').all()

# Good - Select specific columns, eager loading
from sqlalchemy.orm import load_only

articles = Article.query\
    .filter_by(status='published')\
    .options(load_only(
        Article.id, Article.title, Article.slug,
        Article.excerpt, Article.image_url, Article.views
    ))\
    .limit(12).all()

# Good - Avoid N+1 with joinedload
from sqlalchemy.orm import joinedload

parts = Part.query\
    .options(joinedload(Part.zone).joinedload(Zone.segment))\
    .filter_by(status='published').all()

1.4 Add Basic Caching

Cache các queries thường xuyên không đổi

# Simple in-memory cache
from functools import lru_cache
from datetime import datetime, timedelta

# Cache for 5 minutes
cache_timeout = {}

@lru_cache(maxsize=128)
def get_popular_articles(limit=6):
    cache_key = f'popular_{limit}'
    if cache_key in cache_timeout:
        if datetime.now() < cache_timeout[cache_key]:
            # Return cached
            pass

    articles = Article.query.filter_by(status='published')\
        .order_by(Article.views.desc()).limit(limit).all()

    cache_timeout[cache_key] = datetime.now() + timedelta(minutes=5)
    return articles

2.3 Data Migration Script

# scripts/migrate_to_postgres.py
import sqlite3
import psycopg2
from psycopg2.extras import execute_batch

# Export from SQLite
sqlite_conn = sqlite3.connect('unilab.db')
sqlite_conn.row_factory = sqlite3.Row
cursor = sqlite_conn.cursor()

# Connect to PostgreSQL
pg_conn = psycopg2.connect(
    "dbname=unilab user=unilab_user password=secure_password"
)
pg_cursor = pg_conn.cursor()

# Migrate each table
tables = ['vertical', 'segment', 'zone', 'part', 'article', ...]
for table in tables:
    print(f'Migrating {table}...')

    # Get data
    cursor.execute(f'SELECT * FROM {table}')
    rows = cursor.fetchall()

    # Insert to PostgreSQL
    if rows:
        columns = ', '.join(rows[0].keys())
        placeholders = ', '.join(['%s'] * len(rows[0]))
        insert_query = f'INSERT INTO {table} ({columns}) VALUES ({placeholders})'

        execute_batch(pg_cursor, insert_query,
            [tuple(row) for row in rows], page_size=1000)

    pg_conn.commit()
    print(f'✓ Migrated {len(rows)} rows')

pg_conn.close()
sqlite_conn.close()

2.4 Advanced Indexes

# PostgreSQL-specific indexes
from sqlalchemy.dialects.postgresql import TSVECTOR

class Article(db.Model):
    # ... fields ...

    # Full-text search vector
    search_vector = db.Column(TSVECTOR)

    __table_args__ = (
        # B-tree indexes (default)
        Index('idx_article_vertical', 'vertical_slug'),
        Index('idx_article_created', 'created_at', postgresql_using='btree'),

        # GIN index for full-text search
        Index('idx_article_search', 'search_vector', postgresql_using='gin'),

        # Partial index (only published)
        Index('idx_article_published', 'created_at',
            postgresql_where=db.text("status = 'published'")),
    )

# Auto-update search vector with trigger
@db.event.listens_for(Article, 'before_insert')
@db.event.listens_for(Article, 'before_update')
def article_search_vector(mapper, connection, target):
    target.search_vector = db.func.to_tsvector('english',
        f"{target.title} {target.content}")

2.5 Full-Text Search

# Fast search với PostgreSQL
def search_articles(query, page=1, per_page=20):
    results = Article.query.filter(
        Article.search_vector.match(query)
    ).order_by(
        db.func.ts_rank(Article.search_vector,
            db.func.to_tsquery('english', query)).desc()
    ).paginate(page=page, per_page=per_page)

    return results

# 3. Manual cache - Control hoàn toàn
def get_article(slug):
    cache_key = f'article_{slug}'
    article = cache.get(cache_key)

    if article is None:
        article = Article.query.filter_by(slug=slug).first()
        cache.set(cache_key, article, timeout=3600)

    return article

4.2 Index Articles to Elasticsearch

# Create index mapping
es.indices.create(index='articles', body={
    'mappings': {
        'properties': {
            'title': {'type': 'text', 'analyzer': 'standard'},
            'content': {'type': 'text', 'analyzer': 'standard'},
            'excerpt': {'type': 'text'},
            'vertical_slug': {'type': 'keyword'},
            'category': {'type': 'keyword'},
            'tags': {'type': 'keyword'},
            'views': {'type': 'integer'},
            'created_at': {'type': 'date'},
            'image_url': {'type': 'keyword'}
        }
    }
})

# Index article on create/update
@db.event.listens_for(Article, 'after_insert')
@db.event.listens_for(Article, 'after_update')
def index_article_to_es(mapper, connection, target):
    if target.status == 'published':
        es.index(index='articles', id=target.id, document={
            'title': target.title,
            'content': target.content,
            'excerpt': target.excerpt,
            'vertical_slug': target.vertical_slug,
            'category': target.category,
            'tags': target.tags.split(',') if target.tags else [],
            'views': target.views,
            'created_at': target.created_at,
            'image_url': target.image_url
        })

# Search function
def search_articles(query, vertical=None, category=None, page=1, per_page=20):
    must = [
        {'multi_match': {
            'query': query,
            'fields': ['title^3', 'content', 'excerpt^2'],
            'type': 'best_fields'
        }}
    ]

    filters = []
    if vertical:
        filters.append({'term': {'vertical_slug': vertical}})
    if category:
        filters.append({'term': {'category': category}})

    body = {
        'query': {
            'bool': {
                'must': must,
                'filter': filters
            }
        },
        'highlight': {
            'fields': {
                'title': {},
                'content': {'fragment_size': 150}
            }
        },
        'from': (page-1) * per_page,
        'size': per_page,
        'sort': ['_score', {'views': 'desc'}]
    }

    results = es.search(index='articles', body=body)
    return results
